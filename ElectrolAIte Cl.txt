
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.tree import DecisionTreeRegressor
from sklearn.ensemble import ExtraTreesRegressor
from xgboost import XGBRegressor
from lightgbm import LGBMRegressor

# Load the dataset
file_path = r'.xlsx' #Insert your file path here
df = pd.read_excel(file_path)

# Fill missing Electrolyte names by forward filling
df['Electrolyte'] = df['Electrolyte'].ffill()

# Convert comma decimal separator to dot if needed
df = df.replace(',', '.', regex=True).astype({'m (mol/kg)': 'float64', 'gamma': 'float64', 'Osmotic': 'float64'})

# Define models with suppressed warnings for LightGBM
models = {
    'XGB': XGBRegressor(random_state=42, n_estimators=300, learning_rate=0.1, max_depth=5),
}

# Identify unique electrolytes by their virial coefficients
virial_columns = ["Beta(0)", "Beta(1)", "C(phi)", "Beta(2)"]
df['Electrolyte_ID'] = df[virial_columns].astype(str).agg('-'.join, axis=1)
unique_electrolytes = df[['Electrolyte', 'Electrolyte_ID']].drop_duplicates()

# Initialize plotting style
sns.set(style="whitegrid")

# Iterate over each unique electrolyte
for _, (electrolyte, electrolyte_id) in unique_electrolytes.iterrows():
    df_electrolyte = df[df['Electrolyte_ID'] == electrolyte_id].copy()

    # Drop Electrolyte_ID from features
    X = df_electrolyte[['m (mol/kg)']]
    y_gamma = df_electrolyte['gamma']
    y_osmotic = df_electrolyte['Osmotic']

    # Train models and make predictions
    predictions_gamma = []
    predictions_osmotic = []

    for model_name, model in models.items():
        model.fit(X, y_gamma)
        predictions_gamma.append(model.predict(X))

        model.fit(X, y_osmotic)
        predictions_osmotic.append(model.predict(X))

    # Average model predictions
    df_electrolyte['Predicted gamma'] = np.mean(predictions_gamma, axis=0)
    df_electrolyte['Predicted Osmotic'] = np.mean(predictions_osmotic, axis=0)

    # Plot results
    plt.figure(figsize=(8, 6))
    plt.plot(df_electrolyte["m (mol/kg)"], df_electrolyte["Predicted gamma"], label="Predicted γ (Line)", linestyle='-', color='b')
    plt.scatter(df_electrolyte["m (mol/kg)"], df_electrolyte["gamma"], label="Actual γ (Dots)", color='b', marker='o')

    plt.scatter(df_electrolyte["m (mol/kg)"], df_electrolyte["Osmotic"], label="Actual Osmotic (Symbols)", color='r', marker='s')
    plt.plot(df_electrolyte["m (mol/kg)"], df_electrolyte["Predicted Osmotic"], label="Predicted Osmotic (Line)", linestyle='--', color='r')

    plt.xlabel("Molality (mol/kg)")
    plt.ylabel("Coefficients")
    plt.title(f"Predicted vs Actual for {electrolyte}")
    plt.legend()
    plt.show()
